def get_target_for_redirect(target: str, domain: str) -> str:

    import validators

    if validators.url(target):
        target_parsed = validators.url(target)
        domain_parsed = validators.url(f"https://{domain}")

        if target_parsed.netloc.endswith(domain_parsed.netloc):
            return target
        else:
            return domain_parsed.scheme + "://" + domain_parsed.netloc
    else:
        return f"https://{domain}"
